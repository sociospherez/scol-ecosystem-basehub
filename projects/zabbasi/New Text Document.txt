  <h2 id="d-title" class="blue">Pillar B – Data Treatment & Ethics</h2>
  <p>We handle data lawfully, ethically, and transparently to protect privacy and trust.</p>
  <div class="section">
    <h3>Policy Commitments</h3>
    <ul>
      <li><strong>Compliance:</strong> GDPR & UK DPA; Trustees are Data Controllers; IT Lead acts as Processor; DPIAs for high-risk processing.</li>
      <li><strong>Ownership & Access:</strong> Role-based access with least privilege; quarterly permission reviews.</li>
      <li><strong>Security & Integrity:</strong> Encryption in transit/at rest; tested backups; no personal device storage without approval.</li>
      <li><strong>Retention & Minimisation:</strong> Keep only what’s needed; annual clean-ups.</li>
      <li><strong>Ethical Use:</strong> Data supports the mission only; no selling/sharing for marketing; clear privacy notices.</li>
    </ul>
  </div>
  <div class="section kpi">
    <h3>KPIs</h3>
    <ul>
      <li>Zero reportable GDPR breaches</li>
      <li>100% annual data protection training</li>
      <li>Quarterly access reviews logged</li>
      <li>Annual retention review completed</li>
    </ul>
  </div>
</section>

  <h2 id="sec-title" class="red">Pillar C – Security & Risk</h2>
  <p>We safeguard technology, data, and people by anticipating risks and applying strong controls.</p>
  <div class="section">
    <h3>Policy Commitments</h3>
    <ul>
      <li><strong>Access:</strong> MFA on all accounts; least-privilege permissions; quarterly account reviews; immediate leaver revocation.</li>
      <li><strong>Devices:</strong> Up-to-date patches and AV; personal devices require signed security agreement; only approved apps for data.</li>
      <li><strong>Risk:</strong> Maintain an IT Risk Register; quarterly Trustee review; mitigations for high-impact risks.</li>
      <li><strong>Incidents:</strong> Prompt reporting; escalate to Trustees; GDPR reporting within 72 hours if required.</li>
      <li><strong>Continuity:</strong> Regular backups; annual disaster recovery test; backup cover for IT Lead absence.</li>
    </ul>
  </div>

      <li>Successful annual restore test</li>

    // Simple view toggling with focus management and URL hash support
    const views = document.querySelectorAll('.view');
    const home = document.getElementById('home');
    const show = id => {
      views.forEach(v => v.classList.remove('active'));
      const el = document.getElementById(id) || home;
      el.classList.add('active');
      // Move focus to first heading for accessibility
      const h = el.querySelector('h2');
      if(h) h.setAttribute('tabindex','-1'), h.focus();
      // Update hash (without jumping on home)
      if(id === 'home') history.pushState(null, '', location.pathname);
      else history.pushState(null, '', '#' + id);
    };

    // Card clicks
    document.querySelectorAll('.card').forEach(btn=>{
      btn.addEventListener('click', ()=> show(btn.dataset.target));
    });

    // Back buttons
    document.querySelectorAll('.back').forEach(b=>{
      b.addEventListener('click', ()=> show(b.dataset.target));
    });

    // Load by hash if present
    window.addEventListener('load', ()=>{
      const id = location.hash.replace('#','');
      if(id) show(id);
    });

    // Keyboard support: Enter/Space activates card
    document.querySelectorAll('.card').forEach(c=>{
      c.setAttribute('tabindex','0');
      c.addEventListener('keydown', e=>{
        if(e.key === 'Enter' || e.key === ' '){ e.preventDefault(); c.click(); }
      });
    });

    // Handle back/forward navigation
    window.addEventListener('popstate', ()=>{
      const id = location.hash.replace('#','');
      views.forEach(v => v.classList.remove('active'));
      (document.getElementById(id) || home).classList.add('active');
    });
